#!/bin/sh
# cgroup v2 freezer 실측 — 제품 코드가 아니라 제품이 기대는 커널 동작(cgroup.freeze · cgroup.events · 얼린 채 cgroup.kill)을 잰다.
# 비-root: systemd --user 가 위임한 cgroup 안에서 돈다(systemd-run --user -p Delegate=yes).
set -u
D="$HOME/gputeer-freeze-probe"
mkdir -p "$D"
CG="/sys/fs/cgroup$(cut -d: -f3- /proc/self/cgroup)"
echo "self_cgroup=$CG"
mkdir "$CG/runner" "$CG/work" || { echo "PROBE_FAILED mkdir"; exit 2; }
echo $$ > "$CG/runner/cgroup.procs" || { echo "PROBE_FAILED move-self"; exit 2; }
sh -c 'i=0; while :; do i=$((i+1)); echo $i > "$0/counter"; sleep 0.02; done' "$D" &
LOOP=$!
echo $LOOP > "$CG/work/cgroup.procs" || { echo "PROBE_FAILED move-loop"; kill $LOOP; exit 2; }
sleep 0.5
now_ms() { date +%s%3N; }
wait_events() { # $1 = 0|1, 반환: 걸린 ms
  t0=$(now_ms)
  while :; do
    grep -qx "frozen $1" "$CG/work/cgroup.events" && { echo $(( $(now_ms) - t0 )); return 0; }
    [ $(( $(now_ms) - t0 )) -gt 2000 ] && { echo TIMEOUT; return 1; }
    sleep 0.01
  done
}
for round in 1 2 3 4 5; do
  echo 1 > "$CG/work/cgroup.freeze"
  fz=$(wait_events 1)
  a=$(cat "$D/counter"); sleep 1; b=$(cat "$D/counter")
  echo 0 > "$CG/work/cgroup.freeze"
  th=$(wait_events 0)
  sleep 0.5; c=$(cat "$D/counter")
  echo "round=$round freeze_confirm_ms=$fz frozen_counter_before=$a after_1s=$b thaw_confirm_ms=$th after_thaw_0.5s=$c"
done
echo 1 > "$CG/work/cgroup.freeze"; wait_events 1 >/dev/null
echo 1 > "$CG/work/cgroup.kill"
t0=$(now_ms)
while [ -n "$(cat "$CG/work/cgroup.procs")" ]; do
  [ $(( $(now_ms) - t0 )) -gt 5000 ] && { echo "KILL_WHILE_FROZEN=FAILED procs=$(cat "$CG/work/cgroup.procs")"; break; }
  sleep 0.01
done
[ -z "$(cat "$CG/work/cgroup.procs")" ] && echo "KILL_WHILE_FROZEN=OK empty_after_ms=$(( $(now_ms) - t0 ))"
wait $LOOP 2>/dev/null; echo "loop_wait_status=$?"
uname -r
rm -rf "$D"
rmdir "$CG/work" 2>/dev/null
echo PROBE_DONE
