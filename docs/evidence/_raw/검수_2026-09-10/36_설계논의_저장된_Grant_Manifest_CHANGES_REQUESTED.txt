244: struct Spawned {
245:     child: Child,
246:     stdout: thread::JoinHandle<String>,
247:     stderr: thread::JoinHandle<String>,

250: fn drain(child: &mut Child) -> (thread::JoinHandle<String>, thread::JoinHandle<String>) {
251:     let out = child.stdout.take().expect("piped stdout");
252:     let err = child.stderr.take().expect("piped stderr");
253:     (
254:         thread::spawn(move || {
255:             let mut s = String::new();
256:             let _ = BufReader::new(out).read_to_string(&mut s);
257:             s
258:         }),
259:         thread::spawn(move || {
260:             let mut s = String::new();
261:             let _ = BufReader::new(err).read_to_string(&mut s);
262:             s
263:         }),
264:     )
265: }

346: fn finish(mut s: Spawned, label: &str) -> (bool, String) {
347:     let deadline = Instant::now() + Duration::from_secs(60);
348:     loop {
349:         match s.child.try_wait().expect("try_wait") {
350:             Some(status) => {
351:                 let out = s.stdout.join().unwrap_or_default();
352:                 let err = s.stderr.join().unwrap_or_default();
353:                 return (status.success(), format!("{out}{err}"));
354:             }
355:             None if Instant::now() >= deadline => {
356:                 let _ = s.child.kill();
357:                 panic!("{label} 가 60초 안에 끝나지 않았다");
358:             }
359:             None => thread::sleep(Duration::from_millis(50)),
360:         }
361:     }
362: }
